\section{Proofs}
\label{app:proofs}

Propositions~\ref{prop:representation} and \ref{prop:regret} are proved in Section~\ref{sec:theory};
Figure~\ref{fig:theory} illustrates the one-step regret example.

\paragraph{Corollary~\ref{cor:strict}.}
\begin{proof}
Put $Z=f(x_{t+1})$ and $\mathcal C_t=\sigma(z_{\le t},a_{\le t})$.
Because observations and latents are measurable functions of state and nuisance, $\mathcal C_t\subseteq\mathcal H_t$.
Any predictor using a shorter window is also $\mathcal C_t$-measurable.
Conditional least squares and the tower property give
\begin{align*}
\E\|g-Z\|^2
&\ge\E\|Z-\E[Z\mid\mathcal C_t]\|^2\\
&=\E\|Z-\E[Z\mid\mathcal H_t]\|^2
 +\E\|\E[Z\mid\mathcal H_t]-\E[Z\mid\mathcal C_t]\|^2\\
&\ge\E\tr\Cov(Z\mid\mathcal H_t).
\end{align*}
The cross term vanishes because its first factor has conditional mean zero given $\mathcal H_t$.
Take conditionally independent copies $P,P'$ of $p_{t+1}$ given $\mathcal H_t$.
Such copies exist for Euclidean state spaces, using their regular conditional distributions.
The nuisance $b$ and current state $p_t$ are fixed under this conditioning.
The conditional independent-copy variance identity yields
\begin{align*}
\tr\Cov(Z\mid\mathcal H_t)
&=\tfrac12\E[\|f(R(P,b))-f(R(P',b))\|^2\mid\mathcal H_t]\\
&\ge\tfrac{\kappa^2}{2}\E[\|P-P'\|^2\mid\mathcal H_t]\\
&=\kappa^2\tr\Cov(p_{t+1}\mid\mathcal H_t)
=\kappa^2\tr\Sigma_t.
\end{align*}
The final equality follows because $p_t+\mu(a_t)$ is $\mathcal H_t$-measurable.
Taking expectations proves Equation~\ref{eq:innovation}.
Averaging these inequalities gives the corresponding bound for a multi-time prediction objective.
Proposition~\ref{prop:representation} supplies the zero-loss comparator when its assumptions also hold.
\end{proof}

\paragraph{Lemma~\ref{lem:mean}.}
\begin{proof}
Let $m=\E[z'\mid c,a]$.
For every square-integrable measurable $g(c,a)$, expand $g-z'=(g-m)+(m-z')$.
The cross term has expectation zero since $\E[m-z'\mid c,a]=0$.
Hence
\[
\E\|g-z'\|^2=\E\|g-m\|^2+\E\tr\Cov(z'\mid c,a).
\]
The second term does not depend on $g$.
The first is nonnegative and vanishes exactly when $g=m$ almost surely.
The same expansion conditional on $(c,a)$, with $g=z_\star$, proves Equation~\ref{eq:decomposition}.
\end{proof}

\paragraph{The innovation bound.} Corollary~\ref{cor:strict} states
\begin{equation}
\label{eq:innovation}
\E\|g(z_{\le t},a_{\le t})-f(x_{t+1})\|^2
\ge \kappa^2\E\tr\Sigma_t .
\end{equation}

\paragraph{The finite-sample regularizer.}
The finite-sample regularizer differs from Equation~\ref{eq:popsigreg}.
For independent identically distributed samples with characteristic function $\phi$ and empirical estimate $\widehat\phi_n$,
\begin{equation}
\E|\widehat\phi_n(t)-w(t)|^2
=|\phi(t)-w(t)|^2+\frac{1-|\phi(t)|^2}{n}.
\end{equation}
Indeed, $\E\widehat\phi_n=\phi$ and $\E|\widehat\phi_n-\phi|^2=(1-|\phi|^2)/n$, since each complex exponential has modulus one; expanding around $\phi$ gives the identity.
At a Gaussian law, the residual equals $(1-e^{-t^2})/n$.
This is estimator bias, not an optimization floor.
Changing the law changes both terms; dependent temporal samples introduce additional covariance terms.
Finite projections and knots also do not characterize Gaussianity.

\paragraph{Exact ranking under affine dynamics and quadratic costs.}
\begin{proposition}[Exact open-loop ranking under affine dynamics and quadratic costs]
\label{prop:exact}
Fix $z_1$ and deterministic matrices $A,B,Q$, with $Q\succeq0$.
For each deterministic open-loop action sequence, let $z_{t+1}=Az_t+Ba_t+w_t$.
The joint noise law is identical across sequences, has finite second moments, and satisfies $\E w_t=0$.
Temporal independence is not required.
Define $m_1=z_1$, $m_{t+1}=Am_t+Ba_t$, and $J(a_{1:h})=\E\sum_{t\le h}(z_t-z_\star)^\top Q(z_t-z_\star)$.
Then
\begin{equation}
\label{eq:exact}
J(a_{1:h})=\sum_{t\le h}(m_t-z_\star)^\top Q(m_t-z_\star)
+\sum_{t\le h}\tr\big(Q\Cov(z_t)\big).
\end{equation}
The second sum is action-independent, so mean rollouts rank all open-loop sequences exactly.
\end{proposition}
Expected features also suffice when value is linear in them
\citep{wan2019expectation,wan2021expectationcontrol}.
\begin{proof}
Unrolling the two recursions gives
\[
z_t=A^{t-1}z_1+\sum_{k<t}A^{t-1-k}(Ba_k+w_k),\qquad
m_t=A^{t-1}z_1+\sum_{k<t}A^{t-1-k}Ba_k.
\]
Thus $e_t=z_t-m_t=\sum_{k<t}A^{t-1-k}w_k$ has mean zero and finite second moments.
Its law is the same for every deterministic action sequence by the joint-noise assumption.
Consequently $\E z_t=m_t$ and $\Cov(z_t)=\E[e_te_t^\top]$ is action-independent.
For each $t$, expand
\[
(z_t-z_\star)^\top Q(z_t-z_\star)
=(m_t-z_\star)^\top Q(m_t-z_\star)
+2(m_t-z_\star)^\top Qe_t+e_t^\top Qe_t.
\]
The cross term has expectation zero.
The last term has expectation $\tr(Q\E[e_te_t^\top])=\tr(Q\Cov(z_t))$.
Summing proves Equation~\ref{eq:exact}.
Its second sum is the same additive constant for every sequence, so pairwise rankings agree, including ties.
Temporal correlations are already included in $\E[e_te_t^\top]$; no temporal-independence assumption is needed.
\end{proof}

\paragraph{Proposition~\ref{prop:amplitude}.}
\begin{proof}
Partition $\R$ into $I_j=[2\varepsilon j,2\varepsilon(j+1))$, $j\in\mathbb Z$.
Let $q_j=\Pr(G\in I_j)$ for $G\sim\Nrm(0,1)$.
Every $q_j$ is positive and their sum is one.
Let $F_j$ be the conditional Gaussian distribution function on $I_j$, with inverse quantile function $Q_j$ on $(0,1)$.
The conditional CDF is continuous and strictly increasing on the interval interior, so its inverse is measurable and takes values in that interior.

Draw $J$ with probabilities $q_j$, $U\sim\mathrm{Uniform}(0,1)$, and $V\sim\Nrm(0,I_{D-1})$, independently.
Set $b=(J,U,V)$; for $D=1$, omit $V$.
Take $R(p,b)=(p,b)$ and define
\begin{equation}
f_\varepsilon(R(p,b))=\big(Q_J((U+p)/2),V\big).
\end{equation}
The mixture $(U+p_t)/2$ is uniform on $(0,1)$, since $p_t$ is an independent fair bit.
Conditional on $J=j$, the first coordinate therefore has the Gaussian law restricted to $I_j$.
Mixing with weights $q_j$ recovers $\Nrm(0,1)$.
Independence of $V$ gives the joint law $\Nrm(0,I_D)$ and zero population regularizer.

Given the first coordinate $y$, its interval determines $J$ almost surely.
Write $v=F_J(y)$.
Then $p_t=\mathbf1\{v>1/2\}$ and $U=2v-p_t$ almost surely; the other coordinates recover $V$.
Thus the latent recovers both state and nuisance.
For fixed $b$, the two first coordinates are distinct and lie inside the same interval of width $2\varepsilon$.
All remaining coordinates agree, proving the distance bound.

Latent history reveals $b$, while the next bit remains independent and fair.
The next latent consequently has two equiprobable values, say $z^{(0)}(b)$ and $z^{(1)}(b)$.
Lemma~\ref{lem:mean} gives $m(b)=(z^{(0)}(b)+z^{(1)}(b))/2$.
Its conditional squared error is $\|z^{(1)}(b)-z^{(0)}(b)\|^2/4$, strictly positive and bounded by $\varepsilon^2$.
The midpoint is square-integrable by Jensen's inequality, since both state encodings have finite second moments.
Taking expectations proves the loss bound, including strict positivity since the conditional error is positive almost surely.
The midpoint depends only on $b$, so it is independent of $p_{t+1}$.
Finally, decoding the current bit permits $a_t=p_t$.
A state-blind action is independent of the current fair bit, giving success probability $1/2$.
\end{proof}

\begin{remark}[Information must enter the rollout or its score]
\label{rem:voi}
A planner that scores two candidate futures identically when they differ only in a future observation
and the response to it assigns no value to that information; the restriction must hold for its
rollout and its score, terminal value included. An exact closed-loop terminal value can favour retreat
before a reveal even with an open-loop rollout, and replanning can exploit later observations
\citep{howard1966information,barshalom1974dual}. Our oracle search tests a specified horizon and
score; it does not bound all feedback controllers.
\end{remark}

\section{The cube worlds}
\label{app:cubes}

The six cube worlds share the gate's body (frame, cube, actions, border ring, cap, reset distribution
uniform on $[16,42]^2$) without the wall and cross three backgrounds with two dynamics
(Figure~\ref{fig:worlds}). The background is black, independent gray pixel noise redrawn every frame,
or gray noise fixed throughout an episode. Dynamics are deterministic or include an unrendered wind
added to each move; each wind coordinate follows an integer random walk with increments $-1,0,+1$ at
probabilities $0.1,0.8,0.1$, clipped to $[-2,2]$ (mean absolute wind $0.83$ pixels per axis). Fixed
texture is predictable nuisance, fresh texture is unpredictable nuisance, wind changes the controlled
motion. On these worlds the class-token JEPA behaves like the texture optimum of
Proposition~\ref{prop:representation} when the background is fixed per episode (prediction loss
$0.0004$, cube position unreadable, Tables~\ref{tab:models} and \ref{tab:diag}) and the grid JEPA
stays in the noise basin under wind (prediction loss $0.73$ with per-frame noise). Figure~\ref{fig:worlds} shows the
seven worlds and Figure~\ref{fig:gate} three real gate episodes. Table~\ref{tab:cube} gives the
round-1 survival of every controller, Table~\ref{tab:edge} the edge-start re-grading of
Section~\ref{sec:results-rest}, and Table~\ref{tab:sigreg} the regularizer sweep on the fixed-background
still world.

float32 parameters, and one A100. A $500$-step calibration estimates the training length fitting

Model seeds $0$, $1$, $2$ of the grid JEPA differ only in the initialisation and data-order seed.

\paragraph{Flow settings.}
The tokenizer uses $9{,}000$ steps, batch $256$, and learning rate $9\times10^{-4}$; dynamics
$16{,}000$ steps, batch $64$, the same learning rate, width $256$, six layers, eight spatial and
temporal attention heads, a $32$-frame context, independent per-frame noise levels, a shortcut loss of
weight $0.25$ every fourth step after the first quarter of training, history dropout $0.25$, two
history frames resampled every eight steps after half of training, and a temporal loss weight of
$0.25$. The symlog reward head has $255$ bins over $[-20,20]$; reward and termination losses have
weight $0.1$ and enter the trunk at full strength for half of training, then at factor $0.1$.
Inference uses four function evaluations per frame. We corrected terminal-transition masking in the
recipe's reward loss (Appendix~\ref{app:rewardbug}).

\paragraph{JEPA read-outs.}
Post-hoc heads use the frozen encoder and $80{,}000$ frames from a different behaviour-policy seed,
with the last $200$ time rows held out. The pixel decoder uses $8{,}000$ steps, batch $256$, learning
rate $10^{-3}$, width $256$, six residual blocks; it supports the figures and the decoded-latent control, in which its frozen output, clipped to $[0,255]$, replaces the pixels of the real-gate PPO; no true pixel enters that policy after the encoder. Reward,
termination, and behaviour-value heads are two-layer perceptrons of width $512$ on the latent and the
one-hot action, $6{,}000$ steps, batch $1024$, learning rate $10^{-3}$, Monte Carlo value targets at
discount $0.99$, value loss weight $0.1$. The residual predictor models $z_{t+1}-m$ with
$m=g(z_{t-7:t},a_{t-7:t})$ the original grid prediction, conditioned on $m$, $z_t$, and the action:
three residual blocks of width $1024$, $8{,}000$ steps, batch $2048$, $74$ seconds, four function
evaluations per sample. On $20{,}000$ held-out windows the original predictor's squared error is
$3.07\times10^{-4}$, one residual sample $2.05\times10^{-4}$, the mean of eight $1.32\times10^{-4}$.

\paragraph{Policy optimisation.}
PPO uses discount $0.99$, GAE parameter $0.9$, clipping $0.2$, value coefficient $0.5$, entropy
coefficient $0.01$, AdamW at $2.5\times10^{-4}$ with weight decay $0.01$, four epochs per batch, and
gradient clipping at $1.0$. Pixels pass through a three-block Impala torso, latents through a two-layer
perceptron; a GRU receives that embedding and the previous action; the critic uses $64$ categorical
bins with histogram-Gaussian targets. Real-environment PPO uses $128$ environments and $64$ steps per
iteration with four minibatch groups, truncated backpropagation over $16$ steps, and $1.5$M
environment steps. In-model PPO uses $150$ updates of $512$ trajectories of $16$ steps ($1.2$M
imagined steps) with minibatch size $128$; half of the starts are resets, half dataset states with
their real eight-frame context. All reported policy rows use final checkpoints.

\paragraph{Compute.} Logged training times on a shared A100: grid JEPA training $573$ s ($8{,}500$
steps; $1{,}074$ and $3{,}802$ s for the $17{,}000$- and $34{,}000$-step fits); probe and heads $3$

$5$--$6$ min per run at four function evaluations, $20$--$26$ min with the mean of eight; PPO in the real environment $3.7$--$7.7$ min on pixels with the convolutional torso and $51$--$70$ s with the MLP torso, $76$--$78$ s on the frozen latent, $4.7$--$4.8$ min on the decoded latent with the convolutional torso and $1.8$--$2.9$ min with the MLP torso, $1.3$--$3.2$ min on the transformed latents; simulator search at horizon
$8$: $7$ s; the latent diagnostic of Figure~\ref{fig:geometry}: $2$ min.

\section{Evaluation protocol}
\label{app:protocol}

A run fixes a world, a model checkpoint, and a controller. Planning uses $128$ episodes; policies
$256$. Episodes end on death or after $128$ steps. Within each harness, start draws and per-step
environment randomness are shared across controllers. Planning requires seven random warm-up steps to
obtain an eight-frame history, and episodes dying during warm-up are rejected and resampled, which
conditions the planning start distribution on survival; policy and baseline rows start at reset. The
simulator search of Section~\ref{sec:results-ppo} uses the planning harness and its random row is
identical, episode for episode, to the planners' random row. Survival is the fraction alive at the
cap; length is the mean number of steps from reset; walls passed counts walls leaving the frame while
the agent is alive; deaths per $100$ steps is $100$ times deaths divided by total lived steps. Episode
standard errors describe one trained system; training seeds are reported as separate runs.

\paragraph{The latent diagnostic.} From the evaluation set we take $64$ eight-frame contexts whose
last frame has the wall solid at $x\in\{42,45\}$ and reconstruct the simulator state; the recorded
future, its counterfactual with the other gap, and the wall-free future are then rendered with the
recorded actions. Amplitudes are squared distances between the encoder's latents of two rendered
futures at the same step; the one-step residual is the predictor's teacher-forced error on the
recorded future; the gap coordinate is the projection of a rollout latent on the unit vector from the
counterfactual's latent to the recorded one, centred at their midpoint and scaled so that the two
futures sit at $\pm1$. Decoded intensities are the mean minimum colour channel over the wall's pixels
and over each $16$-row gap band.

\paragraph{Craftax.} A grid JEPA trained on Craftax-Classic pixels is graded by six-action goal
reaching from $128$ environments and by reward-MPC over $100$ steps with horizon-six CEM
(Table~\ref{tab:craftax}).

\section{Controls on the latent policy}
\label{app:latentctl}

Table~\ref{tab:latentctl} lists every real-gate PPO run whose observation was built from the frozen JEPA latent or chosen to match it. The $42$ transformed-latent runs use model seeds $0$ and $1$ and three policy seeds each; with the reference arms (pixels with either torso, frozen and decoded latent with the convolutional torso over three model seeds) the table holds $66$ runs, all at $1.5$M steps. The latent policy is an MLP on the flattened observation, the pixel policy the Impala CNN (Appendix~\ref{app:hparams}); the MLP torsos on latents and on flattened frames share the design, not the input width. \emph{Torso.} The MLP torso on the flattened true frame passes $0.84$--$1.00$ walls and survives $2$--$6\%$, the level of the frozen latent, and does not discover the strategy in three runs; on the decoded latent it does the same in six. \emph{Amplified wall signal.} With $z_0$ the encoder's output on the same frame with the wall pixels erased, the policy observes $z + (g-1)(z - z_0)$, the wall's own latent difference scaled $g$-fold (its squared amplitude $g^2$-fold); $g=4$ and $g=16$ produce no discovery. \emph{Subspaces.} A ridge probe (ridge $1$, centred, $80/20$ split of $32{,}000$ random-frame latents of model seed $0$) maps the latent to one-hot wall position ($22$ classes) and gap state (undrawn, top, bottom). Held out, it scores $0.70$ on wall position against a $0.71$ majority rate and $0.90$ on gap state against a $0.90$ majority rate, and on the frames whose gap is drawn it names the side in $0.4\%$ ($0\%$ for model seed $1$): it answers ``undrawn''. Its $22$-dimensional span holds $0.3\%$ of the latent variance against $4.3\%$ for a random span of the same rank. The wall-difference vectors $z-z_0$ of $8{,}192$ frames need $306$ of $512$ directions for $90\%$ of their energy ($295$ for model seed $1$; top direction $1.8\%$); that span carries $90\%$ of the difference energy and $80\%$ of the latent variance, against $60\%$ and $60\%$ for a random span of the same rank. For a span with orthogonal projector $P$ the policy observes $z + (g-1)Pz$ with $g=4$; scaling the probe span, the wall-difference span, or a random span matched to either rank this way produces no discovery.

\section{The reward-head correction in the flow recipe}
\label{app:rewardbug}

The original recipe masks transitions entering reset frames out of reward training. Our worlds assign
their only nonzero reward, $-1$, on terminal transitions, so masking those transitions removes the
entire reward signal: the head predicts zero, reward correlation is undefined, and every policy has
zero discounted return. The corrected head predicts $r_{t+1}$ from the frame-$t$ summary and action
$a_{t+1}$ with terminal transitions kept in its loss; all reported flow models use this form.
